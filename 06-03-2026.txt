1.1. Show department_id and count of employees whose salary is greater than 10000 and job_id contains 'MAN'.

SELECT department_id, COUNT(*)
FROM employees
WHERE salary > 10000
AND job_id LIKE '%MAN%'
GROUP BY department_id;

2.2. Find average salary per department_id for employees hired after 01-JAN-2005 and salary is less than 12000.

SELECT department_id, AVG(salary)
FROM employees
WHERE hire_date > '01-JAN-2005'
AND salary < 12000
GROUP BY department_id;
